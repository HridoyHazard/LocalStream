import path from "node:path";
import { NextRequest, NextResponse } from "next/server";
import { mediaUrlForPath, serverConfig } from "@/lib/server/config";
import {
  FfmpegMissingError,
  getEmbeddedSubtitle,
  listEmbeddedTracks,
  type EmbeddedSubsConfig,
} from "@/lib/server/embedded-subs";

export const dynamic = "force-dynamic";
export const runtime = "nodejs";

/**
 * Subtitles that are embedded inside a video file (typical for MKV and MP4).
 *
 *   GET ?path=<file>             lists the embedded subtitle tracks
 *   GET ?path=<file>&stream=<n>  returns track n as WebVTT. The first request
 *                                starts a background extraction and answers 202;
 *                                poll until it answers 200.
 */

function ffprobeFor(ffmpegPath: string) {
  if (process.env.FFPROBE_PATH) return process.env.FFPROBE_PATH;
  const swapped = ffmpegPath.replace(/ffmpeg(\.exe)?$/i, (_match, ext?: string) => `ffprobe${ext ?? ""}`);
  return swapped === ffmpegPath ? "ffprobe" : swapped;
}

function subsConfig(): EmbeddedSubsConfig {
  return {
    ffmpegPath: serverConfig.ffmpegPath,
    ffprobePath: ffprobeFor(serverConfig.ffmpegPath),
    cacheDir: path.join(path.dirname(serverConfig.databasePath), "subs-cache"),
  };
}

const NO_STORE = { "Cache-Control": "no-store" };

function problem(status: number, message: string, error?: string) {
  return NextResponse.json({ error: error ?? "embedded-subs", message }, { status, headers: NO_STORE });
}

export async function GET(request: NextRequest) {
  const relativePath = request.nextUrl.searchParams.get("path") ?? "";
  const segments = relativePath.split("/");
  if (!relativePath || segments.some((segment) => !segment || segment === "." || segment === ".." || segment.includes("\\"))) {
    return problem(400, "A valid video path is required.");
  }

  const config = subsConfig();
  const url = mediaUrlForPath(relativePath);
  const streamParam = request.nextUrl.searchParams.get("stream");

  try {
    const tracks = await listEmbeddedTracks(config, url);
    if (streamParam === null) {
      return NextResponse.json({ tracks }, { headers: NO_STORE });
    }

    const streamIndex = Number(streamParam);
    const track = Number.isInteger(streamIndex) ? tracks.find((entry) => entry.index === streamIndex) : undefined;
    if (!track) return problem(404, "That subtitle track does not exist in this file.");
    if (!track.supported) {
      return problem(422, `${track.label} is an image based subtitle (${track.codec}) and cannot be shown in the browser.`);
    }

    const result = await getEmbeddedSubtitle(config, relativePath, url, track.index);
    if (result.status === "ready") {
      return new NextResponse(result.vtt, {
        status: 200,
        headers: { "Content-Type": "text/vtt; charset=utf-8", "Cache-Control": "private, max-age=3600" },
      });
    }
    if (result.status === "working") {
      return NextResponse.json({ status: "working", elapsedMs: result.elapsedMs }, { status: 202, headers: NO_STORE });
    }
    return problem(502, `Could not extract this subtitle track: ${result.message}`);
  } catch (error) {
    if (error instanceof FfmpegMissingError) {
      return problem(
        501,
        "Embedded subtitles need ffmpeg. Install it on the computer that runs LocalStream (Windows: winget install Gyan.FFmpeg), then restart LocalStream. If ffmpeg is not on PATH, set FFMPEG_PATH in .env.local.",
        "ffmpeg-missing",
      );
    }
    return problem(502, error instanceof Error ? error.message : "Could not read the embedded subtitles.");
  }
}
